void io_free_rsrc_node(struct io_ring_ctx *ctx, struct io_rsrc_node *node)
{
        if (node->tag)
                io_post_aux_cqe(ctx, node->tag, 0, 0);
        switch (node->type) {
        case IORING_RSRC_FILE:
                fput(io_slot_file(node));
                break;
        case IORING_RSRC_BUFFER:
                io_buffer_unmap(ctx, node->buf);
                break;
        }
}
